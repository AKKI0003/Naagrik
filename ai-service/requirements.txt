# Deliberately CPU-friendly choices — this runs on a free Render/Railway
# instance, not a GPU box. open_clip's smallest ViT-B-32 checkpoint and
# opennsfw2 both run acceptably on CPU for single-image requests.
fastapi==0.115.5
uvicorn[standard]==0.32.0
python-multipart==0.0.17
pillow==11.0.0
torch==2.5.1
# open_clip_torch depends on torchvision — pinning it explicitly to
# the exact release built against torch 2.5.1 stops pip from having
# to search torchvision's whole version history for a match (that
# search is what produced the wall of "torchvision X depends on
# torch Y" lines in a failed build log).
torchvision==0.20.1
open_clip_torch==2.29.0
opennsfw2==0.14.0
numpy==2.1.3
# BLIP captioning (stage 3 — free-text description, not just a
# category label). transformers pulls this model architecture;
# sentencepiece is BLIP's tokenizer dependency.
transformers==4.46.3
sentencepiece==0.2.0
# protobuf is deliberately left unpinned — tensorflow (pulled in
# transitively by opennsfw2/keras) requires protobuf>=6.31.1, and an
# earlier explicit pin here (protobuf==5.28.3, chosen for no reason
# specific to BLIP) caused a hard runtime crash: "Detected mismatched
# Protobuf Gencode/Runtime major versions ... gencode 6.31.1 runtime
# 5.28.3". Letting pip resolve it naturally picks a version that
# satisfies tensorflow's actual constraint.
#
# IMPORTANT — deploy note, not enforceable from this file alone:
# opennsfw2 hard-requires the "opencv-python" package (the GUI build),
# which needs libxcb/libGL and other X11 shared libraries that don't
# exist on headless server images (Railway, Render, most containers).
# `pip install -r requirements.txt` alone WILL crash at import time on
# a headless host with "ImportError: libxcb.so.1: cannot open shared
# object file". Fix: the deploy's Build Command must install this file
# normally, then swap in the headless OpenCV build afterward:
#
#   pip install -r requirements.txt && pip uninstall -y opencv-python && pip install --no-cache-dir opencv-python-headless
#
# (Not something a requirements.txt line can express on its own, since
# opencv-python and opencv-python-headless are two different PyPI
# packages that both happen to provide the same `cv2` import — pip
# will install both if both are merely listed, with no guarantee which
# one's files end up "winning" in site-packages.)
